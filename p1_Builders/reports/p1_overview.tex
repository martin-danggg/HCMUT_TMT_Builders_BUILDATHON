\begin{mdframed}[linecolor=orange,linewidth=1pt,backgroundcolor=orange!8]
\textbf{Điều quan trọng nhất của luật practice:} 15 bot nhà (4 dễ, 4 trung bình, 4 khó, 3 chuyên gia) hành động như \emph{một đội}: chia sẻ tầm nhìn, không đánh nhau, và săn bạn theo kiểu ``một con lúc đầu, thêm dần khi bạn hạ bớt, tất cả vào cuối''. Đề viết ``hunt you weakest first''; mình hiểu là bot yếu được gửi đi trước, nhưng câu này có thể đọc theo cách khác, nên hỏi BTC nếu cần. Vì bot nhà không giết nhau, \textbf{bạn chỉ lên hạng khi bot nhà chết}: do bạn giết, hoặc do vùng bo ở cuối trận (người nhiều HP chết sau). Vòng loại và chung kết thì khác hẳn: đấu tự do với bot của các đội khác. Bot phải mạnh ở cả hai bối cảnh.
\end{mdframed}

\partbox{p1}{1. Yêu cầu của đề}{Image Docker có entrypoint nói protocol \fn{mechfight/1} qua stdin/stdout (JSON từng dòng, tối đa 64 KiB/dòng): trả \fn{ready} trong 20 s, mỗi quan sát trả đúng một action có \fn{tick} trùng, trong 50 ms; khi nhiều quan sát dồn lại thì chỉ trả quan sát mới nhất; thoát sạch sau \fn{end}. Phải qua \fn{check.py} chạy dưới cờ sandbox.}
\partbox{p1}{2. Mục tiêu khi xong}{Repo public riêng \fn{hcmut-tmt-p1-mech-royale} (đã tạo) có \fn{Dockerfile} ở gốc; \fn{check.py} báo PASS khi chạy qua \fn{docker run --network none --cpus 1 --memory 512m --read-only ...}; có bot v1 đã nộp và có điểm.}
\partbox{p1}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} kit (\fn{python/}, \fn{check/}, \fn{arena/}).\\
\textbf{Kiến thức:} Docker, chạy process không phải root, đọc stdin không chặn (wrapper đã có luồng đọc riêng), đo thời gian bằng \fn{time.perf_counter}.}
\partbox{p1}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{python/mechfight.py}, \fn{python/example_bot.py}, \fn{python/Dockerfile} (\fn{FROM python:3.13-slim}), \fn{check/check.py}, \fn{protocol.md} §1--3, §6.\\
\textbf{Đầu ra:} repo \fn{hcmut-tmt-p1-mech-royale/} gồm \fn{Dockerfile}, \fn{mechfight.py}, \fn{bot.py}, \fn{data/} (lưới bản đồ của 1.2), \fn{README.md}.}
\longbox{p1}{5. Hướng giải quyết và các bước làm}{
\begin{enumerate}[leftmargin=1.4em]
  \item Tạo repo public, chép nội dung \fn{python/} lên gốc, đổi \fn{ENTRYPOINT} sang \fn{bot.py}. Nếu dùng numpy thì ghim phiên bản và \fn{pip install} \emph{hệ thống} trong Dockerfile (không \fn{--user}, vì bot chạy với uid 65534).
  \item Chạy \fn{python3 check/check.py -- python3 -u bot.py}, rồi \fn{docker build} và lệnh \fn{check.py --map-dir /arena -- docker run ...} đúng như README của kit.
  \item Thêm bộ đếm thời gian: nếu một quyết định sắp chạm khoảng 30 ms thì trả action rút gọn (giữ di chuyển cũ). Trễ quá 2\% số quan sát là trượt kiểm tra.
  \item Bọc toàn bộ \fn{act()} trong \fn{try/except}: lỗi thì log ra stderr và trả action an toàn (đi về tâm vùng bo), không để process chết (chết là đứng yên làm bia cả trận).
  \item Nộp ngay bản v1 (bot mẫu + đi vào bo an toàn hơn) để kiểm tra quy trình và có mốc điểm.
\end{enumerate}
\trial{Bot mẫu Python qua \fn{check.py} (chạy trực tiếp, không Docker, trên máy chạy thử): PASS, 300/300 quan sát đúng hạn, thời gian trả lời p50 0,3 ms, p99 0,6 ms, tối đa 1,2 ms. Trong container 1 CPU sẽ chậm hơn.}}

Bot qua \fn{check.py} khi chạy thường vẫn có thể trượt khi chạy với cờ sandbox (ghi file ngoài \fn{/tmp}, cài gói cho root). Trả lời sát 50 ms thì điểm có thể khác giữa các lần chạy.}

\begin{enumerate}[leftmargin=1.4em]
  \item Tính ``bo tương lai'': tại thời điểm $t$, tâm và bán kính nội suy giữa vòng hiện tại và vòng kế tiếp; điểm đến phải nằm trong vòng kế tiếp với lề (ví dụ 15 m) để tới nơi trước khi co.
  \item Hàm điểm cho ô ứng viên: nằm trong bo tương lai, gần tường (che chắn ít nhất một phía), không ở giữa bãi trống, xa vị trí địch đã biết, gần cube/vũ khí có thể nhặt.
  \item A* trên lưới 2 m (khoảng 0,2 triệu ô) mỗi 1--2 s, hoặc khi điểm đến đổi; giữa các lần thì theo đường đã có. Gỡ kẹt: nếu vận tốc gần 0 trong 0,5 s thì nhảy + phản lực hoặc đổi hướng.
  \item Hồi máu: không gây và không nhận sát thương 5 s thì hồi 45 HP/s, nên khi HP thấp thì lùi khỏi tầm nhìn địch (đặt tường giữa mình và địch) trong khi vẫn ở trong bo.
  \item Cuối trận (vòng 25 m rồi 0 m, từ khoảng 350 s): vào giữa vòng sớm, giữ HP cao; khi bo về 0, ai còn nhiều HP sống lâu hơn.
  \item Đi im lặng khi cần ẩn: dưới 5 m/s không phát tiếng bước chân.
\end{enumerate}}
\partbox{p3}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Đổi hướng cần thời gian (gia tốc 58,5 m/s² dưới đất, 12 m/s² trên không). Bo không gây sát thương cho drone NPC. Zone damage cũng reset bộ đếm hồi máu 5 s. Đừng để bot nhà ``săn'' mình ở bãi trống: bot nhà chia sẻ tầm nhìn, một con thấy là cả đội biết.}
\metabox{p3}{25\% của P1 (6,5\% tổng), khoảng 1,6 giờ}{1.2}{1.4, 1.5}{Bỏ qua (làm một mình)}

\subsection{1.4 Chiến đấu và vũ khí}
\partbox{p4}{1. Yêu cầu của đề}{Mọi phát bắn đi theo hướng thân \emph{thực tế} (thân quay dần về hướng lệnh với 4 rad/s), không có khóa mục tiêu; súng bay thẳng 252 m/s nên phải ngắm đón; đạn trúng nhà, cầu, địa hình đúng chỗ nhìn thấy; trúng từ phía sau nhân 1,5.}
\partbox{p4}{2. Mục tiêu khi xong}{Bot bắn trúng mục tiêu di chuyển ở 50--150 m, không phí đạn vào tường, quay mặt về phía kẻ đang bắn mình, dùng đúng vũ khí theo khoảng cách và theo vũ khí của đối thủ.}
\partbox{p4}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} \fn{obs.mechs} (vị trí, vận tốc, hướng ngắm, \fn{charging}, vũ khí), \fn{projectiles}, \fn{beams}, \fn{events}, \fn{self.weapons}.\\
\textbf{Kiến thức:} bài toán đón đầu (giải phương trình bậc hai cho thời gian bay), quỹ đạo parabol cho cối (vận tốc 72 m/s, trọng lực 22 m/s²), tam giác vũ khí của đề.}
\partbox{p4}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{protocol.md} §5 (kích thước, gốc đạn), §7 (vũ khí, giáp sau, tam giác vũ khí).\\
\textbf{Đầu ra:} \fn{combat.py} (chọn mục tiêu, ngắm, bắn), \fn{weapons.py} (luật riêng cho súng, cối, laser, sét).}
\longbox{p4}{5. Hướng giải quyết và các bước làm}{
\textbf{Ý tưởng:} một bộ ``ngắm'' chung cộng luật riêng cho từng vũ khí; chỉ bóp cò khi sai số hướng nhỏ và \fn{los} thông.
\begin{enumerate}[leftmargin=1.4em]
  \item Ngắm đón cho súng: với địch ở $\mathbf{p}$, vận tốc $\mathbf{v}$, tìm $t$ nhỏ nhất thỏa $\|\mathbf{p}+\mathbf{v}t-\mathbf{m}\| = 252\,t$ (từ nòng $\mathbf{m}$), ngắm vào tâm khối \fn{pos.z + 4.5}.
  \item Chỉ bắn khi \fn{self.aim} đã gần hướng lệnh (sai số góc nhỏ hơn khoảng $\arctan(2{,}5/d)$) và tia từ nòng đến điểm đón không vướng nhà.
  \item Chọn mục tiêu: điểm số theo HP còn lại thấp, khoảng cách gần, đang bắn mình, có tầm bắn thông. Ưu tiên kết liễu (mỗi mạng hạ được là một hạng khi mình còn sống).
  \item Quay mặt: khi nhận \fn{damaged} với \fn{rear: true}, xoay thân về \fn{bearing}. Thân và chân độc lập, nên vừa lùi về chỗ che vừa bắn trả được.
  \item Né: đi ngang vuông góc với hướng địch; thấy địch \fn{charging} laser thì đổi hướng ngang hoặc nấp (laser bắn theo hướng ngắm lúc nhả cò).
  \item Laser: giữ cò 0,75 s rồi nhả khi đã ngắm trúng; tầm 280 m, 315 sát thương, hợp đánh xa. Sét: chỉ dùng khi địch dưới 45 m trong nón 30°, tốn năng lượng. Cối: bắn vòng qua nhà theo tiếng động hoặc vị trí nhớ, tính góc nâng cho tầm cần.
  \item Nạp đạn khi không có mục tiêu (nạp chủ động), đừng để hết băng giữa giao tranh.
\end{enumerate}}
\partbox{p4}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Cối nổ gần mình thì mình cũng mất 50\% sát thương lan và bị hất văng. Đang nạp laser thì đi chậm một nửa. Cầm cối mà bị laser bắn thì nhận thêm 60\% (cook-off). Sét bắn vào khoảng không vẫn tốn năng lượng. Đạn của mình không hiện trong \fn{projectiles}; sự kiện \fn{hit} cho biết trúng hay không để tự hiệu chỉnh.}
\metabox{p4}{25\% của P1 (6,5\% tổng), khoảng 1,6 giờ}{1.2, 1.3}{1.5, 1.6}{Bỏ qua (làm một mình)}

  \item Nâng cấp, ưu tiên sống sót: \fn{plating} (+15\% HP) $>$ \fn{armor} ($-$8\% sát thương nhận) $>$ \fn{nanites} (hồi máu) $>$ \fn{gun_damage}/\fn{laser_damage} $>$ phần còn lại. Luôn trả \fn{pick} ngay, không để mặc định.
  \item Supply drop (3 cube + đầy HP + đầy năng lượng): chỉ tranh khi vòng drop không có địch nhìn thấy và mình đến trước; nếu không thì tránh vì đó là nơi mọi bot kéo đến.
