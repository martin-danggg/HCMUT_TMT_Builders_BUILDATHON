\subsection{Nguồn thông tin và quy ước số liệu}
\begin{itemize}
  \item \textbf{Nguồn chính:} trang đề (ảnh chụp). \fn{challenge.pdf} (3 trang) dùng cho phần yêu cầu kỹ thuật, trừ các điểm trang đề đã ghi đè. Repo chưa có forum Q\&A.
  \item \textbf{Tham khảo:} \fn{upload.py} (16 luồng mặc định, mảnh 1 MiB, thử lại 4 lần, bỏ qua file đã có cùng kích thước và MD5 \emph{trong cùng thư mục đích}).
  \item Chưa có số liệu chạy thử: mình không truy cập được \fn{machina-input.joinbuilders.org} từ môi trường làm việc (bị chặn mạng), nên chưa biết tỉ lệ từng định dạng. Bước đầu của 3.1 là lấy \fn{manifest.tsv} và đếm.
  \item Số liệu ``quan sát'': ổ D: chứa repo có 272 GB, còn trống khoảng 113 GB (đo từ máy của bạn qua kết nối desktop). \textbf{Không đủ} cho 139,7 GB đầu vào + file nén giải ra + đầu ra.
\end{itemize}

\partbox{p1}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} trang đề mục ``1. Get the input'', \fn{get-input.sh --help}.\\
\textbf{Đầu ra:} \fn{work/manifest.tsv}, \fn{work/inventory.parquet} (đường dẫn, kích thước, phần mở rộng, magic, sha256), \fn{notes/p3_inventory.md}.}

  \item \fn{curl -fsSLO https://machina-input.joinbuilders.org/get-input.sh}; \fn{bash get-input.sh --limit 100}; lấy \fn{manifest.tsv}.
  \item Thống kê từ manifest (chưa cần tải hết): phần mở rộng, thư mục, dung lượng. Chọn mẫu đại diện cho mỗi định dạng để làm 3.3.

\subsection{3.2 Nhận diện danh tính và bảng bút danh}
\partbox{p2}{1. Yêu cầu của đề}{Thay tên công ty và người, tài khoản, thông tin liên lạc, tên miền, địa chỉ, logo, tên và số hiệu dự án/chi tiết mang tính định danh. Cùng một danh tính phải nhận cùng một bút danh trên toàn corpus; danh tính khác nhau phải giữ khác nhau; trường định danh phải điền bút danh có nghĩa, không để trống.}
\partbox{p2}{2. Mục tiêu khi xong}{Danh sách danh tính (mỗi danh tính có loại, các biến thể viết) và hàm sinh bút danh tất định, có cùng độ dài byte với tên gốc khi cần, không trùng nhau. Bảng ánh xạ lưu riêng tư, ngoài thư mục nộp.}
\partbox{p2}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} metadata từ mọi định dạng (PDF Info/XMP, thuộc tính OLE, \fn{docProps}, EXIF, header STEP \fn{FILE_NAME}), tên thư mục và file, văn bản trích từ PDF/bản vẽ (khung tên: công ty, người vẽ, người kiểm).\\
\textbf{Kiến thức:} regex cho email, URL, tên miền, số điện thoại; HMAC để sinh bút danh tất định; NER (spaCy) chỉ để gợi ý; mã hóa UTF-8/UTF-16LE.}
\partbox{p2}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{work/inventory.parquet}, kết quả trích metadata của 3.3.\\
\textbf{Đầu ra:} \fn{code/identities.py} (luật nhận diện), \fn{code/pseudonym.py} (sinh bút danh), \fn{private/mapping.json} (\emph{không} nộp).}
\longbox{p2}{5. Hướng giải quyết và các bước làm}{
\textbf{Ý tưởng:} danh sách danh tính = (thu thập tự động từ các trường chắc chắn là định danh) + (regex cho liên lạc) + (rà tay các ứng viên có tần suất cao).
\begin{enumerate}[leftmargin=1.4em]
  \item Trích mọi trường metadata kiểu tác giả, công ty, người lưu cuối, bản quyền, tổ chức; gom giá trị khác nhau, đếm tần suất.
  \item Regex: email, URL, tên miền (đuôi phổ biến), số điện thoại; tách tên miền từ email để thay nhất quán.
  \item Tên thư mục cấp cao thường là tên khách hàng/dự án: liệt kê và đánh dấu.
  \item Mẫu số hiệu chi tiết/dự án (ví dụ chữ + số theo một định dạng lặp lại): suy ra định dạng từ dữ liệu, sinh bút danh \emph{giữ nguyên khuôn dạng} (chữ thay bằng chữ, số thay bằng số, giữ dấu gạch).
  \item Biến thể: hoa/thường, viết tắt, có/không có dấu cách hay gạch dưới, dạng sở hữu, bản UTF-16. Tất cả biến thể của một danh tính ánh xạ về cùng một bút danh (giữ kiểu hoa thường của nơi xuất hiện).
  \item Bút danh có nghĩa và tất định: \fn{HMAC(secret, danh tính chuẩn hóa)} chọn từ danh sách tên giả (công ty, người, đường phố), tên miền dùng miền dành riêng như \fn{example.com}; chỉnh độ dài cho bằng tên gốc; kiểm tra không có hai danh tính trùng bút danh.
\end{enumerate}}
\partbox{p2}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Không thay nhầm thông tin kỹ thuật không định danh: tên vật liệu, đơn vị, dung sai, tên tiêu chuẩn (ISO, ASME), tên phần mềm (``SolidWorks'', ``Autodesk Inventor'' trong trường Producer/Application là phần mềm, không phải danh tính). Tên ngắn hay tên trùng từ thường (ví dụ một họ người trùng một từ tiếng Anh) dễ thay nhầm: chỉ thay khi khớp nguyên từ và có ngữ cảnh.}
\metabox{p2}{20\% của P3 (5,2\% tổng), khoảng 1,3 giờ}{3.1}{3.3, 3.4}{Bỏ qua (làm một mình)}

Đổi nội dung PDF có thể làm hỏng font subset (chữ thay không có glyph): kiểm tra hiển thị trên mẫu. File trên 2 GB không được chấm. Không giữ file tạm hay bảng ánh xạ trong thư mục nộp.}

\subsection{3.4 Đổi tên file, thư mục và cập nhật tham chiếu}
\partbox{p4}{1. Yêu cầu của đề}{Tên file và thư mục cũng phải ẩn danh. Khi đổi tên file thì đổi luôn các tham chiếu tới nó. Giữ phần mở rộng và các phần đường dẫn không định danh để scorer tìm được file nó kiểm tra.}
\partbox{p4}{2. Mục tiêu khi xong}{Bảng đổi tên (đường dẫn gốc $\rightarrow$ đường dẫn mới) không trùng lặp; mọi tham chiếu (assembly $\rightarrow$ part, bản vẽ $\rightarrow$ model, xref DWG, liên kết trong PDF/HTML, đường dẫn trong file nén) trỏ đúng tên mới.}
\partbox{p4}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} danh sách đường dẫn (manifest), bảng bút danh, các chuỗi tham chiếu tìm thấy ở 3.3.\\
\textbf{Kiến thức:} cách assembly lưu đường dẫn tới part (thường là đường dẫn tuyệt đối/tương đối dạng chuỗi trong file), giới hạn 400.000 object.}
\partbox{p4}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{work/inventory.parquet}, \fn{private/mapping.json}.\\
\textbf{Đầu ra:} \fn{code/rename.py}, \fn{private/rename_map.json} (không nộp).}
\longbox{p4}{5. Hướng giải quyết và các bước làm}{
\begin{enumerate}[leftmargin=1.4em]
  \item Tách mỗi đường dẫn thành các phần; chỉ thay token định danh bên trong mỗi phần (tên công ty, dự án, người, số hiệu), giữ phần còn lại và phần mở rộng.
  \item Dùng bút danh cùng độ dài để đường dẫn mới dài bằng đường dẫn cũ: tham chiếu trong file nhị phân được sửa tại chỗ bằng phép thay cùng độ dài ở 3.3.
  \item Kiểm tra trùng: hai đường dẫn khác nhau không được ra cùng tên mới (kể cả khi so không phân biệt hoa thường).
  \item Áp bảng đổi tên cho cả đường dẫn bên trong file nén.
  \item Ghi lại các tham chiếu không sửa được (ví dụ nằm trong luồng nén của SolidWorks) để đưa vào README.
\end{enumerate}}
\partbox{p4}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Đổi tên mà quên đổi tham chiếu làm assembly mất liên kết, có thể mất cả nhóm check ``quan hệ giữa file''. Đổi cả phần không định danh của đường dẫn (ví dụ thư mục \fn{drawings/}) làm scorer không tìm thấy file. README phải dùng đường dẫn \emph{mới} hoặc mô tả trung tính, không nhắc tên gốc.}
\metabox{p4}{10\% của P3 (2,6\% tổng), khoảng 0,65 giờ}{3.2, 3.3}{3.5}{Bỏ qua (làm một mình)}

\subsection{3.5 Kiểm định và đóng gói attempt}
\partbox{p5}{1. Yêu cầu của đề}{Mỗi attempt có \fn{corpus/} (file đã ẩn danh), \fn{code/} (toàn bộ mã nguồn và phiên bản thư viện hoặc lockfile), \fn{README.txt} (lệnh chạy chính xác, giới hạn, ngoại lệ, lỗi sẵn có của nguồn). Không đưa thông tin đăng nhập, dữ liệu thô, file tạm, log riêng tư. Upload bằng \fn{upload.py} với token 1 giờ.}
\partbox{p5}{2. Mục tiêu khi xong}{Một attempt hợp lệ đã upload; bộ quét rò rỉ báo 0 lần xuất hiện tên gốc; mọi file mở được hoặc được ghi rõ là lỗi sẵn có của nguồn.}
\partbox{p5}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} đầu ra của 3.3--3.4, danh sách tên gốc (riêng tư).\\
\textbf{Kiến thức:} tìm chuỗi nhiều mẫu nhanh (Aho--Corasick, \fn{ripgrep}), cách \fn{upload.py} nối tiếp khi hết token (\fn{--token-file}).}
\partbox{p5}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{upload.py}, trang đề mục 3--4.\\
\textbf{Đầu ra:} \fn{attempt-N/corpus/}, \fn{attempt-N/code/}, \fn{attempt-N/README.txt}; \fn{code/validate.py}, \fn{code/leakscan.py}.}
\longbox{p5}{5. Hướng giải quyết và các bước làm}{
\begin{enumerate}[leftmargin=1.4em]
  \item Kiểm tra mở file theo định dạng (\fn{pikepdf.open}, \fn{PIL.Image.verify}, \fn{zipfile.testzip}, \fn{olefile}, \fn{ezdxf.readfile}, đọc header STEP). So với file gốc: file gốc mở được mà đầu ra không mở được là lỗi nghiêm trọng, phải quay về bản gốc.
  \item Quét rò rỉ trên mọi byte đầu ra, cả bên trong file nén và luồng zlib đã giải nén, với các dạng ASCII, UTF-16LE, không phân biệt hoa thường, URL-encode.
  \item Đếm object (tối đa 400.000) và file lớn hơn 2 GB.
  \item Viết README: lệnh chạy, phiên bản, giới hạn, danh sách file không xử lý được kèm lý do (dùng đường dẫn mới).
  \item Upload: \fn{export MACHINA_UPLOAD_TOKEN=...}; \fn{python3 upload.py --dest} \fn{gs://builders-machina-submissions-1120/}\allowbreak\fn{teams/<team-id>/attempt-N/} \fn{--corpus ./cleaned-corpus --code ./code} \fn{--readme ./README.txt}; hết token thì lấy token mới và dán khi được hỏi; chạy lại sẽ bỏ qua file đã lên.
  \item Dán địa chỉ thư mục \fn{attempt-N/} (kết thúc bằng \fn{/}) vào form.
\end{enumerate}}
\partbox{p5}{6. Yêu cầu đặc biệt, chỗ dễ sai}{\fn{upload.py} chỉ bỏ qua file đã có \emph{trong cùng thư mục đích}, nên mỗi attempt mới phải upload lại toàn bộ khoảng 140 GB; với mạng gia đình đây là giới hạn thực sự của số attempt (nên hỏi BTC có cách sao chép phía server không). Không sửa attempt đã nộp cho tới khi công bố kết quả. Chỉ dùng thư mục của đội mình.}
\metabox{p5}{15\% của P3 (3,9\% tổng), khoảng 1,0 giờ (chưa kể upload nền)}{3.3, 3.4}{3.6}{Bỏ qua (làm một mình)}
