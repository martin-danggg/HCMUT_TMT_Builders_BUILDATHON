\subsection{Bảng điểm và cách chấm}
P4 chỉ có một tiêu chí, \textbf{chấm tự động hoàn toàn} bằng scorer (cùng luật với \fn{score.py} trong kit). Không có phần chấm AI. Có một yếu tố \textbf{người chấm}: BTC ghi rõ ``Top entries are read'', tức code của các đội đứng đầu sẽ được người đọc để kiểm tra luật (không có bảng tra patch $\rightarrow$ test, không chia sẻ code giữa đội).

\partbox{p1}{2. Mục tiêu khi xong}{Có repo nộp riêng; \fn{make sample} chạy được và in điểm; có môi trường offline giống hệt lúc BTC gán nhãn (cùng image, cùng \fn{requirements.lock}, \fn{SKIP_INTEGRATION=0}) để 4.2 dựng map đúng.}
\partbox{p1}{3. Dữ liệu đầu vào và kiến thức cần có}{

\textbf{Kiến thức:} Docker (buildx, \fn{--platform linux/amd64}, quyền file cho uid 65534), pytest 9 (subtest, \fn{--collect-only}), cách scorer đọc stdout (bảng mã lỗi trong \fn{SCORING.md}).}
\partbox{p1}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{Dockerfile} (giữ nguyên digest \fn{python:3.13-slim@sha256:e48f59...}), \fn{Makefile}, \fn{score.py}, \fn{requirements.lock} (43 gói, gồm pytest 9.2.0, pytest-xdist 3.8.0, pytest-timeout 2.4.0, duckdb 2.0.0).\\

\longbox{p1}{5. Hướng giải quyết và các bước làm}{
\begin{enumerate}[leftmargin=1.4em]
  \item Tạo repo public mới, chép toàn bộ nội dung kit vào gốc (kit không có thư mục cấp cao nhất). Kiểm tra \fn{make sample} cho ra \fn{Simulated: 1210.470 s}, \fn{Misses: 0} (đúng với lần chạy thử của mình).

\subsection{4.2 Map tĩnh từ commit ghim}
\partbox{p2}{1. Yêu cầu của đề}{Đề cho phép ``maps built from the pinned commit (coverage, imports, fixtures)'' và cấm ``bảng tra từ patch hoặc từ loại thay đổi sang test''. Map phải dựng từ mã sqlglot tại \fn{eb11aec}, không dựng từ \fn{answers.json}.}
\partbox{p2}{2. Mục tiêu khi xong}{Có 3 map nén, tổng dưới vài MB, commit vào repo nộp: (a) \textbf{coverage}: với mỗi file \fn{sqlglot/**.py}, dòng $\rightarrow$ tập số hiệu test đã chạy qua dòng đó; (b) \textbf{fixture}: file dữ liệu dưới \fn{tests/} $\rightarrow$ tập test đã mở file đó; (c) \textbf{import}: module $\rightarrow$ tập file test import nó (trực tiếp hoặc gián tiếp).}
\partbox{p2}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} môi trường offline của 4.1, \fn{tests.json} (để đổi node id sang số hiệu).\\
\textbf{Kiến thức:} coverage.py \emph{dynamic context} (\fn{--cov-context=test} của pytest-cov, hoặc \fn{dynamic_context = test_function}), \fn{coverage json --show-contexts}, \fn{sys.addaudithook} (sự kiện \fn{open}) để ghi file mà mỗi test mở, \fn{ast} để tìm phạm vi hàm/lớp.}
\partbox{p2}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} bản checkout sqlglot, \fn{sample/tests.json}.\\
\textbf{Đầu ra:} \fn{maps/coverage.json.gz}, \fn{maps/fixtures.json.gz}, \fn{maps/imports.json.gz}, \fn{maps/funcs.json.gz} (phạm vi dòng của từng hàm/lớp), và script dựng map \fn{tools/build_maps.py} (commit để giải thích được khi BTC đọc code).}
\longbox{p2}{5. Hướng giải quyết và các bước làm}{
\textbf{Ý tưởng:} chạy toàn bộ test một lần với coverage theo ngữ cảnh từng test, rồi nén lại.
\begin{enumerate}[leftmargin=1.4em]
  \item Trong môi trường offline, cài thêm \fn{coverage} và \fn{pytest-cov} (chỉ dùng để dựng map, không đưa vào image nộp).
  \item Chạy \fn{SKIP_INTEGRATION=0 pytest --cov=sqlglot --cov-context=test -p no:randomly -n 4}. Tốn khoảng 3--5 lần thời gian gốc (103 s một luồng). Ước lượng 2--5 phút với 4 worker xdist (chưa đo).
  \item Thêm \fn{conftest} plugin nhỏ: trong hook \fn{pytest_runtest_call}, gắn \fn{sys.addaudithook} ghi lại mọi đường dẫn mở dưới \fn{tests/} kèm node id $\rightarrow$ map fixture.
  \item Map import: với từng file test, chạy \fn{python -c "import tests.x"} trong process mới và lấy \fn{sys.modules} (hoặc \fn{python -X importtime}). Lưu ý \fn{sqlglot/__init__.py} import gần như cả package, nên thường mọi file test phụ thuộc mọi module.
  \item Đổi node id sang số hiệu theo \fn{tests.json}; với test có subtest, gộp context về id gốc.
  \item Nén: nhiều dòng có cùng tập test, nên lưu ``bảng tập test duy nhất'' + ``dòng $\rightarrow$ chỉ số tập''. Ghi kèm \fn{funcs.json.gz}: cho mỗi file, danh sách (đầu dòng, cuối dòng, tên) của hàm/lớp lấy từ \fn{ast}.
  \item Tự kiểm: với 30 commit mẫu, tập test ``phủ dòng bị đổi'' phải chứa mọi test trong \fn{failing} (chỉ dùng để \emph{kiểm tra} map, không dùng để tạo luật).
\end{enumerate}}
\partbox{p2}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Dòng chạy lúc import (định nghĩa lớp, hằng số, dict \fn{TRANSFORMS} trong dialect) bị \emph{mọi} test phủ, nên map coverage ở các dòng này vô dụng: cần xử lý riêng ở 4.3. Docstring không phải dòng thực thi, coverage không thấy; nếu sqlglot có test chạy doctest thì phải map docstring sang test đó. Test lấy chậm nhất: \fn{test_max_nodes} 28,817 s, \fn{test_execute_tpcds} 21,454 s, \fn{test_generator_memory} 6,686 s. Đừng dựng map trong \fn{Dockerfile}: mỗi lần nộp sẽ mất vài phút build và dễ vượt 10 phút.}
\metabox{p2}{30\% của P4 (6,6\% tổng), khoảng 1,6 giờ}{4.1}{4.3, 4.5}{Bỏ qua (làm một mình)}

\subsection{4.3 Phân tích patch và chọn test}
\partbox{p3}{1. Yêu cầu của đề}{Mỗi patch là một unified diff độc lập so với commit ghim. In tập test có thể fail; tập rỗng (\fn{"workers": []}) là đáp án đúng cho thay đổi không làm hỏng gì.}
\partbox{p3}{2. Mục tiêu khi xong}{Hàm \fn{select(patch) -> set[int]} cho từng loại file bị sửa, chạy vài mili-giây mỗi commit, không sót test fail nào trên 30 commit mẫu và tập chọn nhỏ hơn hẳn ``chạy hết''.}
\partbox{p3}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} các map của 4.2, file patch.\\
\textbf{Kiến thức:} định dạng unified diff (\fn{@@ -a,b +c,d @@}, dòng \fn{-}/\fn{+}/ngữ cảnh), \fn{compile()} để kiểm tra cú pháp, cấu trúc sqlglot (\fn{expressions}, \fn{parser}, \fn{generator}, \fn{dialects/}, \fn{optimizer/}, \fn{tests/fixtures/optimizer/*.sql}).}
\partbox{p3}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{/data/commits/*.patch}, \fn{maps/*.json.gz}, bản sqlglot trong image (clone tại commit ghim trong \fn{Dockerfile}, được phép) để áp patch và kiểm tra cú pháp.\\
\textbf{Đầu ra:} module \fn{select.py} trong repo nộp.}
\longbox{p3}{5. Hướng giải quyết và các bước làm}{
\textbf{Ý tưởng:} patch so với commit ghim, nên số dòng phía \fn{-} chính là số dòng trong map. Ánh xạ dòng cũ bị đổi sang test qua map, mở rộng ra mức hàm để không sót nhánh mới.
\begin{enumerate}[leftmargin=1.4em]
  \item Tự viết parser diff: với mỗi file, lấy các dòng cũ bị xóa/sửa; với dòng chỉ thêm mới, lấy dòng cũ ngay trước và ngay sau điểm chèn.
  \item \fn{sqlglot/**.py}, dòng nằm trong một hàm: lấy mọi test phủ \emph{bất kỳ dòng nào của hàm đó} (mức hàm). Mức dòng chính xác hơn nhưng sót khi patch thêm nhánh \fn{elif} mới mà chưa test nào đi qua.
  \item Dòng ở mức lớp/module (thuộc tính lớp, entry trong dict \fn{TRANSFORMS}, \fn{FUNCTIONS}, hằng số): coverage nói ``mọi test'', nên dùng luật hẹp hơn: test phủ các hàm của cùng lớp, cộng các test trong file test của dialect đó. Nếu không chắc thì dùng 4.5.
  \item Áp patch vào bản sao file trong \fn{/tmp} và gọi \fn{compile()}. Lỗi cú pháp làm file không import được, nên mọi test import module đó đều fail: chọn theo map import (thường là toàn bộ).
  \item \fn{tests/fixtures/**}: chọn các test đã mở file đó (map fixture).
  \item \fn{tests/**.py}: chọn test có thân hàm bị đổi; nếu đổi ở mức module (helper, import) thì chọn cả file test; \fn{conftest.py} thì chọn mọi test trong phạm vi của nó.
  \item Patch chỉ thêm comment: trong \fn{.py} thì gần như vô hại, \emph{trừ khi} rơi vào docstring (có thể làm hỏng doctest) hoặc file \fn{.sql} fixture (comment là một phần dữ liệu test). Đề nói rõ ``comment không được hứa là vô hại'', nên không cho comment đi tắt sang tập rỗng.
  \item File khác (\fn{pyproject.toml}, \fn{setup.cfg}, \fn{Makefile}, tài liệu): không có trong mẫu. Cấu hình build/test thì chọn tất cả; tài liệu thuần (\fn{.md} ngoài \fn{tests/}) thì chọn rỗng.
\end{enumerate}}
\partbox{p3}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Patch có thể sửa nhiều file (loại \fn{multi}). Đổi tên hoặc xóa file (\fn{/dev/null}) phải xử lý được. Một commit mẫu (\fn{3utz}) làm fail 37 test trên 4 file khác nhau, nên đừng chỉ nhìn file test cùng tên. Mọi luật phải suy ra từ mã nguồn hoặc map; ví dụ luật ``fixture $\rightarrow$ \fn{test_optimizer.py}'' phải đến từ map fixture chứ không phải từ việc thấy trong \fn{answers.json}.}
\metabox{p3}{25\% của P4 (5,5\% tổng), khoảng 1,4 giờ}{4.2}{4.5, 4.6}{Bỏ qua (làm một mình)}

\subsection{4.4 Chia worker tối ưu}
\partbox{p4}{1. Yêu cầu của đề}{Tối đa 8 worker sau khi bỏ worker rỗng. Chi phí là tải worker lớn nhất cộng 2 s mỗi worker; thời lượng test tính bằng mili-giây nguyên.}
\partbox{p4}{2. Mục tiêu khi xong}{Hàm \fn{split(tests) -> workers} cho chi phí nhỏ nhất (hoặc rất sát) với mọi tập test.}
\partbox{p4}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} tập test từ 4.3, \fn{seconds} trong \fn{tests.json}.\\
\textbf{Kiến thức:} bài toán xếp lịch máy song song (makespan), thuật toán LPT, quy hoạch động/brute force cho tập nhỏ.}
\partbox{p4}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} hàm \fn{split} trong \fn{selector.py} của starter (LPT, cố định 3 worker).\\
\textbf{Đầu ra:} \fn{split.py} chọn số worker $k$ tối ưu.}
\longbox{p4}{5. Hướng giải quyết và các bước làm}{
\textbf{Ý tưởng:} với mỗi $k=1..8$, chia bằng LPT rồi chọn $k$ nhỏ nhất theo $\text{makespan}_k + 2000k$ (ms).
\begin{enumerate}[leftmargin=1.4em]
  \item Đổi \fn{seconds} sang ms bằng \fn{round(s*1000)}, giống scorer.
  \item Với mỗi $k$, chạy LPT (test dài nhất trước, vào worker nhẹ nhất). Thêm bước cải thiện cục bộ (đổi chỗ 1--1 giữa worker nặng nhất và nhẹ nhất).
  \item Tập rỗng thì in \fn{[]}. Tập rất nhỏ (dưới khoảng 12 test) thì thử hết cách chia để ra tối ưu thật.
  \item Cận dưới để tự kiểm: $\max(\max_t d_t,\ \lceil\sum d_t/k\rceil) + 2000k$.
\end{enumerate}
\trial{Với toàn bộ 1620 test: $k=3$ cho 40,349 s/commit (đúng starter), $k=4$ cho \textbf{36,817 s}, $k=5$ cho 38,817 s, $k=8$ cho 44,817 s. Bỏ 3 test chậm nhất (tổng còn 46,09 s): $k=5$ cho 19,218 s. Nghĩa là quyết định có chọn \fn{test_max_nodes} (28,8 s) và \fn{test_execute_tpcds} (21,5 s) hay không quan trọng hơn cách chia.}}
\partbox{p4}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Nhiều worker hơn chưa chắc tốt hơn: mỗi worker thêm 2 s, và một test 28,8 s không chia nhỏ được. Không lặp số hiệu test trong cùng một dòng (\fn{repeated_test}). Số phải là số nguyên JSON, không phải chuỗi.}

\subsection{4.5 Luật an toàn (recall)}
\partbox{p5}{1. Yêu cầu của đề}{Final có ``loại thay đổi không có trong practice''. Một commit sót test fail mất 300 s. BTC sửa nhãn sai nếu phát hiện trước khi công bố kết quả.}
\partbox{p5}{2. Mục tiêu khi xong}{Bộ luật dự phòng rõ ràng: khi map không đủ chắc chắn thì mở rộng tập test theo bậc (hàm $\rightarrow$ lớp $\rightarrow$ file test liên quan $\rightarrow$ toàn bộ), với ngưỡng dựa trên kỳ vọng chi phí.}
\partbox{p5}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} kết quả 4.3 kèm ``độ tin cậy'' (dòng thuộc hàm hay thuộc module, có lỗi cú pháp không, loại file).\\
\textbf{Kiến thức:} kỳ vọng chi phí: thêm test tốn tối đa khoảng 36,8 s (chạy hết, 4 worker), còn sót tốn 300 s.}
\partbox{p5}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{select.py}, map import, 30 commit mẫu.\\
\textbf{Đầu ra:} \fn{policy.py} và một bảng luật ngắn trong README của repo nộp (để giải thích khi BTC đọc code).}
\longbox{p5}{5. Hướng giải quyết và các bước làm}{
\textbf{Ý tưởng:} chọn tất cả khi xác suất sót ngoài tập chọn vượt khoảng $36{,}8/300 \approx 12\%$; nếu không thì giữ tập hẹp.
\begin{enumerate}[leftmargin=1.4em]
  \item Liệt kê các tình huống ``mù'' của map: dòng mức module/lớp, file \fn{.py} mới thêm, file không phải Python, đổi tên file, xóa hàm, đổi chữ ký hàm, đổi decorator, đổi \fn{__init__.py}.
  \item Gán cho mỗi tình huống một bậc mở rộng cố định (không dựa vào \fn{answers.json}): ví dụ xóa hoặc đổi tên hàm thì thêm mọi test phủ \emph{nơi gọi} hàm đó (tìm bằng \fn{ast} trên bản sqlglot trong image).
  \item Lỗi cú pháp hoặc lỗi import mức module thì chọn tất cả (nếu cần chắc chắn có thể chạy \fn{python -c "import sqlglot"} với file đã vá trong \fn{/tmp}, tốn khoảng nửa giây CPU, chỉ làm khi đổi mức module).
  \item Kiểm tra recall: chạy lại 30 commit mẫu, rồi tự tạo vài patch kiểu khác (đổi tên hàm, thêm file, sửa docstring) và chạy thật pytest offline để xem luật có bắt được không. Đây là kiểm thử, không phải bảng tra.
  \item Giữ CPU thấp: nạp map một lần, xử lý mọi patch trong một process.
\end{enumerate}}
\partbox{p5}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Đừng tối ưu quá sát bộ mẫu (30 commit) hay bảng practice (khoảng 40 commit): final có loại thay đổi mới. Không dùng bảng ``loại thay đổi $\rightarrow$ test'' kể cả viết tay. Test có subtest: một subtest fail là cả test fail (pytest 9 có thể in \fn{PASSED} cạnh \fn{SUBFAILED}).}
\metabox{p5}{15\% của P4 (3,3\% tổng), khoảng 0,8 giờ}{4.2, 4.3}{4.6}{Bỏ qua (làm một mình)}
