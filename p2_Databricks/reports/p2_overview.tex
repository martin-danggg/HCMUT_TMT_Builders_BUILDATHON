\subsection{2.2 Đọc CSV đúng dialect}

\textbf{Kiến thức:} \fn{read_csv} của DuckDB với \fn{delim}, \texttt{quote=\textquotesingle{}\textquotesingle{}}, \texttt{escape=\textquotesingle{}\textquotesingle{}}, \fn{header}, \texttt{new\_line=\textquotesingle{}\textbackslash n\textquotesingle{}}; hàm chuỗi \fn{string_split}, \fn{regexp_matches} (dùng \fn{[0-9]}, không dùng \texttt{\textbackslash d}).}
\partbox{p2}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{rules.md} mục ``CSV dialect'', \fn{fixtures/README.md}.\\

\longbox{p2}{5. Hướng giải quyết và các bước làm}{
\textbf{Ý tưởng:} không để DuckDB hiểu dấu phẩy hay ngoặc. Đọc mỗi dòng thành một chuỗi, rồi tự tách.
\begin{enumerate}[leftmargin=1.4em]
  \item Đọc nguyên dòng: \texttt{read\_csv(glob, columns=\{\textquotesingle{}line\textquotesingle{}:\textquotesingle{}VARCHAR\textquotesingle{}\}, delim=chr(1), quote=\textquotesingle{}\textquotesingle{}, escape=\textquotesingle{}\textquotesingle{}, header=true, auto\_detect=false, new\_line=\textquotesingle{}\textbackslash n\textquotesingle{})}. Header bị bỏ theo từng file; dòng giống header ở giữa file vẫn là dữ liệu (fixture L3).
  \item Tách: \texttt{f = string\_split(line, \textquotesingle{},\textquotesingle{})}; giữ dòng có \fn{len(f) = ncols}. Dòng trống cho 1 trường nên tự bị loại khi \fn{ncols} $\ge$ 2.
  \item Với mỗi trường: nếu không chứa \fn{"} thì giữ nguyên; nếu có dạng \fn{"..."} (ngoặc ở đầu và cuối, bên trong không có ngoặc) thì bỏ ngoặc; còn lại là dòng lỗi. Trường rỗng (sau khi bỏ ngoặc) thành NULL.
  \item Không trim, không đổi hoa thường; \texttt{\textbackslash r} ở cuối trường cuối giữ nguyên để các bước sau tự loại (giá \texttt{1000\textbackslash r} sai, \fn{ts} có \texttt{\textbackslash r} sai).
  \item Kiểm tra từng fixture dialect bằng cách in bảng thô và so với mô tả trong \fn{fixtures/README.md}.
\end{enumerate}}

Ký tự phân cách giả \fn{chr(1)} phải không xuất hiện trong dữ liệu (đề không cấm hẳn; nên đọc \fn{generator/src/text.rs} để chắc, hoặc thêm kiểm tra). Malformed chiếm khoảng 0,05\% dòng.}

\subsection{2.3 Timestamp và kiểm tra hợp lệ}
\partbox{p3}{1. \subsection{2.4 Gộp ID (linked IDs)}
\partbox{p4}{1. Yêu cầu của đề}{Mỗi dòng \fn{links} hợp lệ (không lỗi, hai phía đều khác rỗng) nối hai ID thành một người; quan hệ bắc cầu, không phụ thuộc thứ tự, file hay chiều. Tên người là ID nhỏ nhất theo byte; ID không xuất hiện trong link nào là một người riêng. Link không có thời gian và áp dụng trước mọi bước khác.}
\partbox{p4}{2. Mục tiêu khi xong}{Bảng \fn{person_map(user_id, person)} cho mọi ID xuất hiện trong \fn{links}; ID khác ánh xạ về chính nó. Qua L1--L4 và A12.}
\partbox{p4}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} \fn{links_raw} từ 2.2.\\
\textbf{Kiến thức:} union-find (gộp theo hạng, nén đường đi) hoặc \fn{scipy.sparse.csgraph.connected_components}; so sánh chuỗi theo byte (thứ tự code point của Python trùng với thứ tự byte UTF-8).}
\partbox{p4}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{rules.md} mục ``Linked IDs'', phần mô tả generator (chuỗi hàng trăm link, thiết bị dùng chung gộp hàng nghìn người).\\
\textbf{Đầu ra:} hàm \fn{build_person_map()} và bảng DuckDB \fn{person_map}.}
\longbox{p4}{5. Hướng giải quyết và các bước làm}{
\textbf{Ý tưởng:} đồ thị làm bằng Python (hoặc scipy), không dùng CTE đệ quy (chuỗi dài hàng trăm bước sẽ rất chậm).
\begin{enumerate}[leftmargin=1.4em]
  \item Lấy các cặp từ DuckDB ra Arrow/numpy; mã hóa ID thành số nguyên bằng \fn{factorize} (hoặc \fn{dense_rank} trong SQL).
  \item Tìm thành phần liên thông (scipy) hoặc union-find thuần Python.
  \item Với mỗi thành phần, tên = ID nhỏ nhất theo byte (lấy \fn{min} trên chuỗi).
  \item Đưa \fn{person_map} trở lại DuckDB, \fn{LEFT JOIN} vào events và clicks: \fn{coalesce(person, user_id)}.
\end{enumerate}}
\partbox{p4}{6. Yêu cầu đặc biệt, chỗ dễ sai}{ID chữ hoa (\fn{U1}) là ID khác, xếp trước mọi ID chữ thường và có thể thành tên người. Link tự nối (\fn{u6,u6}) và link lặp không đổi gì. ID có thể được nối mà không xuất hiện trong event hay click nào, nhưng vẫn có thể là tên (L2). Dòng link có một phía rỗng bị bỏ, và việc đó có thể tách một người làm hai.}
\metabox{p4}{10\% của P2 (2,6\% tổng), khoảng 0,65 giờ}{2.2}{2.5, 2.6}{Bỏ qua (làm một mình)}

% ---------------------------------------------------------------
\subsection{2.5 Event: dedup, lọc, chia phiên}
\partbox{p5}{1. Yêu cầu của đề}{Theo đúng thứ tự: (1) parse, bỏ dòng lỗi và \fn{event_id} rỗng; (2) dedup theo \fn{event_id}, giữ bản có \fn{ingest_ts} nhỏ nhất, \emph{trước} khi kiểm tra hợp lệ; (3) kiểm tra bản giữ lại; (4) bỏ event trễ khi \fn{ingest_ts - ts} quá 24 giờ; (5) chia phiên theo người, sắp theo \fn{ts}, khoảng cách quá 30 phút (1.800.000 ms) thì sang phiên mới, các event cùng \fn{ts} chung phiên; (6) ngày của phiên là ngày UTC lúc bắt đầu; (7) danh mục lấy từ \fn{products.csv}, không có thì \fn{unknown}.}
\partbox{p5}{2. Mục tiêu khi xong}{Bảng \fn{sessions(person, session_id, start_ts, day, events, revenue_cents, has_purchase)} và bảng doanh thu theo danh mục mỗi phiên. Qua E1--E20.}
\partbox{p5}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} \fn{events_raw}, \fn{person_map}, \fn{products.csv}.\\
\textbf{Kiến thức:} window function (\fn{lag}, \fn{sum() over}), \fn{arg_min}/\fn{QUALIFY row_number()}, phép chia ngày theo ms.}
\partbox{p5}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{rules.md} mục ``Events, in this order''.\\
\textbf{Đầu ra:} các bước SQL \fn{events_kept}, \fn{events_valid}, \fn{sessions} trong pipeline.}
\longbox{p5}{5. Hướng giải quyết và các bước làm}{
\begin{enumerate}[leftmargin=1.4em]
  \item Dedup: \fn{QUALIFY row_number() OVER (PARTITION BY event_id ORDER BY ingest_ms) = 1} (các bản cùng \fn{ingest_ts} giống hệt nhau nên chọn bản nào cũng được). Làm trên bảng thô đã parse, chưa lọc hợp lệ.
  \item Lọc hợp lệ (2.3), rồi bỏ trễ: giữ khi \fn{ingest_ms - ts_ms <= 86.400.000} (đúng 24 giờ vẫn giữ; \fn{ingest_ts} trước \fn{ts} vẫn giữ).
  \item Gán người: \fn{person = coalesce(person_map.person, user_id)}.
  \item Chia phiên trên các \emph{timestamp khác nhau}: lấy \fn{DISTINCT person, ts}, tính \fn{gap = ts - lag(ts)}, cờ phiên mới khi \fn{gap > 1.800.000} hoặc đầu tiên, \fn{session_no = sum(cờ) over (...)}; rồi join ngược về event theo \fn{(person, ts)}.
  \item Phiên: \fn{start_ts = min(ts)}, \fn{day = ngày UTC của start_ts}, đếm event, tổng giá các \fn{purchase}, có mua hay không.
  \item Danh mục: \fn{LEFT JOIN products}, null thành \fn{unknown}; tổng giá mua theo (người, ngày, danh mục) để tìm \fn{top_category} ở 2.7.
\end{enumerate}}
\partbox{p5}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Đúng 30 phút vẫn cùng phiên, 30 phút + 1 ms thì tách (E1, E2). Bản sao sớm nhất có \fn{ts} sai thì cả event bị bỏ, bản sau hợp lệ không cứu được (E5). Bản sao có thể mang user khác: lấy user của bản sớm nhất (E4). Phiên qua nửa đêm tính cho ngày bắt đầu (E3). Event trễ phải bỏ \emph{trước} khi chia phiên, vì nó có thể nối hai phiên.}
\metabox{p5}{15\% của P2 (3,9\% tổng), khoảng 1,0 giờ}{2.3, 2.4}{2.6, 2.7}{Bỏ qua (làm một mình)}

\subsection{2.6 Click: gán phiên và đếm hỗ trợ}
\partbox{p6}{1. Yêu cầu của đề}{Click: parse, bỏ \fn{click_id} rỗng, dedup theo \fn{click_id} (các bản giống nhau), kiểm tra hợp lệ. Mỗi phiên được gán cho click hợp lệ của cùng người có \fn{click.ts <= session_start} và cách không quá 7 ngày (604.800.000 ms), lấy \fn{ts} muộn nhất, hòa thì \fn{click_id} nhỏ nhất theo byte; không có thì gán \fn{organic}. Hỗ trợ: mọi chiến dịch có ít nhất một click hợp lệ của người đó trong \fn{[start - 7 ngày, start]}, mỗi chiến dịch tính một lần, kể cả chiến dịch được gán; \fn{organic} không bao giờ hỗ trợ.}
\partbox{p6}{2. Mục tiêu khi xong}{Mỗi phiên có \fn{campaign_id} được gán và danh sách chiến dịch hỗ trợ. Qua A1--A12, L4.}
\partbox{p6}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} \fn{clicks_raw}, \fn{person_map}, \fn{campaigns.csv}, bảng \fn{sessions}.\\
\textbf{Kiến thức:} \fn{ASOF JOIN} của DuckDB, range join (IEJoin), \fn{DISTINCT} theo cặp.}
\partbox{p6}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} \fn{rules.md} mục ``Clicks''.\\
\textbf{Đầu ra:} bảng \fn{session_credit(session, campaign_id, channel)}, \fn{session_assist(session, campaign_id)}.}
\longbox{p6}{5. Hướng giải quyết và các bước làm}{
\begin{enumerate}[leftmargin=1.4em]
  \item Dedup click: \fn{DISTINCT ON (click_id)} sau khi parse (bản sao chỉ khác cách đặt ngoặc, đã bỏ ở 2.2).
  \item Lọc hợp lệ, gán người như event.
  \item Gán phiên: rút gọn click về một dòng cho mỗi \fn{(person, ts)} giữ \fn{min(click_id)} (đúng luật hòa), rồi \fn{ASOF JOIN} theo \fn{person} và \fn{click.ts <= start_ts}; giữ khi \fn{start_ts - click.ts <= 604.800.000}, ngược lại là \fn{organic}.
  \item Hỗ trợ: rút gọn về \fn{(person, campaign_id, ts)} khác nhau; range join với phiên theo \fn{person} và \fn{start_ts - 7 ngày <= ts <= start_ts}; \fn{DISTINCT (session, campaign_id)}.
  \item Tự kiểm: tổng \fn{assisted_sessions} $\ge$ số phiên được gán cho chiến dịch khác \fn{organic}.
\end{enumerate}}
\partbox{p6}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Click đúng lúc bắt đầu phiên được tính, trễ 1 ms thì không (A4). Click giữa phiên chỉ ảnh hưởng các phiên sau (A9). Click trước cửa sổ event tới 7 ngày vẫn tính (A2, A10). Một click gán được nhiều phiên và hỗ trợ mọi phiên trong 7 ngày sau nó, mỗi phiên tính theo ngày của chính phiên đó (A8, A12). Chiến dịch chỉ hỗ trợ mà không được gán vẫn có dòng, các cột gán bằng 0 (A11). Khách ``cá voi'' có rất nhiều click: range join phải dùng kiểu join bất đẳng thức, không được thành tích Descartes.}
\metabox{p6}{15\% của P2 (3,9\% tổng), khoảng 1,0 giờ}{2.3, 2.4, 2.5}{2.7}{Bỏ qua (làm một mình)}

\subsection{2.8 Tối ưu tốc độ}
\partbox{p8}{1. Yêu cầu của đề}{Điểm là thời gian thực từ lúc container khởi động đến khi thoát, gồm cả khởi động engine và đọc CSV. Practice khoảng 20 triệu event; final khoảng 60 triệu, mỗi lần tối đa 3 phút.}
\partbox{p8}{2. Mục tiêu khi xong}{Bản đúng chạy dưới 60 s ở practice (yêu cầu tối thiểu theo đề), mục tiêu thực tế vài giây với DuckDB; nếu viết lại bằng Rust thì dưới 2 s.}
\partbox{p8}{3. Dữ liệu đầu vào và kiến thức cần có}{
\textbf{Dữ liệu:} bộ dữ liệu lớn tự tạo: \fn{generate --seed 7 --events 20000000 --clicks 5000000} (tỉ lệ click giống dev; ước tính khoảng 1,8 GB, chưa đo).\\
\textbf{Kiến thức:} \fn{EXPLAIN ANALYZE} của DuckDB, \fn{threads}, \fn{memory_limit}, \fn{preserve_insertion_order=false}; với Rust: mmap, rayon, bảng băm, sắp xếp song song.}
\partbox{p8}{4. File cần dựa vào $\rightarrow$ file đầu ra}{
\textbf{Dựa vào:} pipeline đúng của 2.7.\\
\textbf{Đầu ra:} pipeline nhanh hơn, bảng đo thời gian trong \fn{notes/p2_log.md}.}
\longbox{p8}{5. Hướng giải quyết và các bước làm}{
\begin{enumerate}[leftmargin=1.4em]
  \item Đo từng bước bằng \fn{EXPLAIN ANALYZE} trên bộ 20 triệu tự tạo; thường tốn nhất là đọc + tách chuỗi, dedup event và chia phiên (sắp xếp theo người, thời gian).
  \item Giảm số lần quét: tách trường và parse timestamp trong cùng một lượt; chỉ giữ các cột cần.
  \item Thay regex bằng hàm chuỗi rẻ hơn (\fn{length}, \fn{substr}, so ký tự) ở đường nóng; chỉ dùng regex khi chuỗi có ký tự lạ.
  \item Mã hóa \fn{person} thành số nguyên trước khi sắp xếp và join (chuỗi chậm hơn nhiều).
  \item Tránh Python ở đường nóng; chỉ phần union-find chạy Python, trên mảng số.
  \item Nếu còn từ 4 giờ trở lên sau khi bản DuckDB đã đúng: viết lại bằng Rust (đọc file song song theo đoạn, tách byte thủ công, union-find, sắp xếp theo người), so khớp với bản DuckDB trên nhiều seed trước khi nộp.
\end{enumerate}}
\partbox{p8}{6. Yêu cầu đặc biệt, chỗ dễ sai}{Chỉ 2 lượt/giờ: luôn kiểm tra đúng ở máy trước khi nộp. Thời gian practice dao động tới khoảng 20\% giữa các máy. Lần chạy ẩn không cho xem log, chỉ báo số dòng thiếu/thừa/khác. Một lần final quá 3 phút là mất điểm final, nên giữ dư địa lớn cho dữ liệu gấp 3.}
\metabox{p8}{15\% của P2 (3,9\% tổng), khoảng 1,0 giờ}{2.7}{Kết quả cuối của P2}{Bỏ qua (làm một mình)}
